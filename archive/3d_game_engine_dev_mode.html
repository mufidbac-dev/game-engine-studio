Script Python Blender: Buat Terrain & Karakter (.GLB)

Buka aplikasi Blender, masuk ke tab Scripting, klik New, tempelkan kode di bawah ini lalu klik Run Script. Script ini akan otomatis membuat Terrain Berbukit dan Karakter Robot/Lowpoly beserta Armature siap di-export ke .glb!

import bpy
import math

# 1. Bersihkan Scene Awal
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# 2. Buat Terrain (Grid + Displace Procedural)
bpy.ops.mesh.primitive_grid_add(x_subdivisions=64, y_subdivisions=64, size=60, location=(0, 0, 0))
terrain = bpy.context.active_object
terrain.name = "Game_Terrain"

# Tambahkan Material Hijau Terrain
mat_terr = bpy.data.materials.new(name="Terrain_Mat")
mat_terr.use_nodes = True
mat_terr.node_tree.nodes["Principled BSDF"].inputs["Base Color"].default_value = (0.12, 0.38, 0.18, 1)
terrain.data.materials.append(mat_terr)

# Buat Gelombang Bukit
for v in terrain.data.vertices:
    v.co.z = math.sin(v.co.x * 0.15) * math.cos(v.co.y * 0.15) * 2.5

# Shade Smooth Terrain
bpy.ops.object.shade_smooth()

# 3. Buat Karakter Game (Capsule / Robot Stylized)
bpy.ops.mesh.primitive_uv_sphere_add(radius=0.6, location=(0, 0, 1.2))
char_body = bpy.context.active_object
char_body.name = "Player_Character"

# Buat Kepala / Visor
bpy.ops.mesh.primitive_cube_add(size=0.5, location=(0, 0.35, 1.5))
visor = bpy.context.active_object
visor.scale = (0.8, 0.4, 0.3)

# Gabungkan Kepala ke Badan Karakter
bpy.ops.object.select_all(action='DESELECT')
visor.select_set(True)
char_body.select_set(True)
bpy.context.view_layer.objects.active = char_body
bpy.ops.object.join()

# Material Karakter Biru Menyala
mat_char = bpy.data.materials.new(name="Character_Mat")
mat_char.use_nodes = True
mat_char.node_tree.nodes["Principled BSDF"].inputs["Base Color"].default_value = (0.15, 0.55, 0.95, 1)
char_body.data.materials.append(mat_char)

print("Terrain & Karakter berhasil dibuat! Silakan File -> Export -> glTF 2.0 (.glb)")
